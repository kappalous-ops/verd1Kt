def gradient_animated(text, offset=0):
    if not text:
        return text
    
    result = []
    length = len(text)
    
    base_colors = [
        (0, 20, 180),
        (0, 80, 220),
        (30, 150, 255),
        (100, 200, 255),
        (200, 230, 255),
        (255, 255, 255)

    for i, char in enumerate(text):
        if char == '\n':
            result.append(char)
            continue
        
        pos = i / length if length > 0 else 0
        wave = math.sin(pos * math.pi * 4 + offset) * 0.5 + 0.5
        
        idx = wave * (len(base_colors) - 1)
        idx1 = int(math.floor(idx))
        idx2 = min(idx1 + 1, len(base_colors) - 1)
        frac = idx - idx1
        
        c1 = base_colors[idx1]
        c2 = base_colors[idx2]
        
        pulse = math.sin(offset * 0.5) * 0.1 + 0.9
        
        r = int((c1[0] + (c2[0] - c1[0]) * frac) * pulse)
        g = int((c1[1] + (c2[1] - c1[1]) * frac) * pulse)

        b = min(255, max(0, b))
        
        color_code = 16 + (36 * (r // 51)) + (6 * (g // 51)) + (b // 51)
        result.append(f"\033[38;5;{color_code}m{char}")
    
    return ''.join(result) + '\033[0m'
